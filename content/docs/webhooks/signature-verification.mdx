title: Signature verification
description: Verify that every webhook delivery came from Rehabify and wasn't altered, using HMAC-SHA256 over the timestamp and raw body. Includes samples for Node.js, Python, PHP and Go.
---

Every delivery is signed with your endpoint's signing secret: `whsec_` followed by 40 alphanumeric characters. Each environment has its own secret, and you can reveal it in the Partner Portal under **Developers → Webhooks**. Always verify the signature before you trust a payload.

## The signature header

```http
X-Rehabify-Signature: t=1791884113,v1=94cb1b4033aa43ffa747a13629a033766fe033d8f82f8c17664ef1bf075dae4c
```

| Part | Meaning |
| :--- | :--- |
| `t` | When the delivery was signed, in Unix seconds. It's also sent as `X-Rehabify-Timestamp`. |
| `v1` | Hex-encoded HMAC-SHA256 signature. |

## Algorithm

```text
v1 = hex( HMAC_SHA256( key = signing secret, message = "<t>.<raw request body>" ) )
```

1. Read the **raw** request body as bytes, exactly as received. Don't parse and re-serialize the JSON: any change in whitespace or key order breaks the signature.
2. Split `X-Rehabify-Signature` on `,`, and each part on the first `=`, to get `t` and `v1`.
3. Reject the delivery if `t` is more than **300 seconds** away from your current time, in either direction. This blocks replayed requests.
4. Build the signed message: the value of `t`, a literal `.`, then the raw body.
5. Compute HMAC-SHA256 of that message, using the full signing secret (including `whsec_`) as the key, and hex-encode the result.
6. Compare your result with `v1` in **constant time**. If they don't match, reject the delivery with a `4xx` status.

Use the secret of the environment the event came from. Sandbox events are signed with your sandbox secret, and live events with your live secret. Each delivery attempt, including retries, is signed when it's sent, so `t` is always recent.

## Test vector

Use these values to unit-test your implementation. The body is shortened for readability, and the secret is an example, not a real one.

| Input | Value |
| :--- | :--- |
| Secret | `whsec_ExampleSecretForDocsOnly0000000000000000` |
| `t` | `1791884113` |
| Raw body | `{"id":"evt_01JA4T7Q9Y2V3W4X5Z6A7B8C9D","object":"event","type":"ping"}` |
| Signed message | `1791884113.{"id":"evt_01JA4T7Q9Y2V3W4X5Z6A7B8C9D","object":"event","type":"ping"}` |
| Expected `v1` | `94cb1b4033aa43ffa747a13629a033766fe033d8f82f8c17664ef1bf075dae4c` |

When you test with this vector, skip the 300-second timestamp check, because `t` is in the past.

```js
// webhook-server.mjs (Node.js 18+). Install with: npm install express
import crypto from "node:crypto";
import express from "express";

const SECRET = process.env.REHABIFY_WEBHOOK_SECRET; // whsec_… for this environment
const TOLERANCE_SECONDS = 300;

export function verifyRehabifySignature(rawBody, header, secret, now = Math.floor(Date.now() / 1000)) {
  if (!header || !Buffer.isBuffer(rawBody)) return false;

  let timestamp = null;
  const signatures = [];
  for (const part of header.split(",")) {
    const i = part.indexOf("=");
    if (i === -1) continue;
    const key = part.slice(0, i).trim();
    const value = part.slice(i + 1).trim();
    if (key === "t") timestamp = value;
    else if (key === "v1") signatures.push(value);
  }
  if (!timestamp || !/^\d+$/.test(timestamp) || signatures.length === 0) return false;
  if (Math.abs(now - Number(timestamp)) > TOLERANCE_SECONDS) return false;

  const expected = crypto
    .createHmac("sha256", secret)
    .update(`${timestamp}.`)
    .update(rawBody)
    .digest(); // 32-byte Buffer

  return signatures.some((sig) => {
    const received = Buffer.from(sig, "hex");
    // timingSafeEqual throws a RangeError if the lengths differ, so check them first.
    return received.length === expected.length && crypto.timingSafeEqual(received, expected);
  });
}

const app = express();

// express.raw() keeps the exact bytes that were signed.
// Register this route BEFORE any app.use(express.json()), or the raw body is lost.
app.post("/webhooks/rehabify", express.raw({ type: "application/json" }), (req, res) => {
  if (!verifyRehabifySignature(req.body, req.get("X-Rehabify-Signature"), SECRET)) {
    return res.status(400).send("invalid signature");
  }

  const event = JSON.parse(req.body.toString("utf8"));
  // 1. Skip events whose id you've already processed.
  // 2. Queue the work, and ignore stale events using data.sequence.
  console.log("Verified", event.id, event.type);
  res.sendStatus(200); // respond within 10 seconds
});

app.listen(3000);
```

```python
# app.py: pip install flask
import hashlib
import hmac
import json
import os
import time

from flask import Flask, abort, request

SECRET = os.environ["REHABIFY_WEBHOOK_SECRET"]  # whsec_... for this environment
TOLERANCE_SECONDS = 300


def verify_rehabify_signature(raw_body, header, secret, now=None):
    if not header:
        return False
    timestamp = None
    signatures = []
    for part in header.split(","):
        key, sep, value = part.strip().partition("=")
        if not sep:
            continue
        if key == "t":
            timestamp = value
        elif key == "v1":
            signatures.append(value)
    if not timestamp or not timestamp.isdigit() or not signatures:
        return False
    now = time.time() if now is None else now
    if abs(now - int(timestamp)) > TOLERANCE_SECONDS:
        return False
    expected = hmac.new(
        secret.encode("utf-8"),
        timestamp.encode("ascii") + b"." + raw_body,
        hashlib.sha256,
    ).hexdigest()
    # compare_digest on bytes is constant-time and safe for unequal lengths
    return any(hmac.compare_digest(expected.encode("ascii"), sig.encode("utf-8")) for sig in signatures)


app = Flask(__name__)


@app.post("/webhooks/rehabify")
def rehabify_webhook():
    raw_body = request.get_data()  # raw bytes: verify before parsing
    if not verify_rehabify_signature(raw_body, request.headers.get("X-Rehabify-Signature"), SECRET):
        abort(400)
    event = json.loads(raw_body)
    # Skip already-processed event ids, then queue the work.
    print("Verified", event["id"], event["type"])
    return "", 200
```

```php
<?php
// webhook.php (PHP 7.4+)
const TOLERANCE_SECONDS = 300;

function verify_rehabify_signature(string $rawBody, string $header, string $secret, ?int $now = null): bool
{
    $timestamp = null;
    $signatures = [];
    foreach (explode(',', $header) as $part) {
        $pair = explode('=', trim($part), 2);
        if (count($pair) !== 2) {
            continue;
        }
        if ($pair[0] === 't') {
            $timestamp = $pair[1];
        } elseif ($pair[0] === 'v1') {
            $signatures[] = $pair[1];
        }
    }
    if ($timestamp === null || !ctype_digit($timestamp) || count($signatures) === 0) {
        return false;
    }
    $now = $now ?? time();
    if (abs($now - (int) $timestamp) > TOLERANCE_SECONDS) {
        return false;
    }
    $expected = hash_hmac('sha256', $timestamp . '.' . $rawBody, $secret);
    foreach ($signatures as $signature) {
        if (hash_equals($expected, $signature)) { // constant-time
            return true;
        }
    }
    return false;
}

$secret = getenv('REHABIFY_WEBHOOK_SECRET'); // whsec_... for this environment
$rawBody = file_get_contents('php://input');  // raw body: don't re-encode $_POST
$header = $_SERVER['HTTP_X_REHABIFY_SIGNATURE'] ?? '';

if (!is_string($secret) || !verify_rehabify_signature($rawBody, $header, $secret)) {
    http_response_code(400);
    exit('invalid signature');
}

$event = json_decode($rawBody, true);
// Skip already-processed event ids, then queue the work.
http_response_code(200);
echo json_encode(['received' => true]);
```

```go
// main.go (Go 1.18+)
package main

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"io"
	"log"
	"net/http"
	"os"
	"strconv"
	"strings"
	"time"
)

const toleranceSeconds = 300

func verifyRehabifySignature(rawBody []byte, header, secret string, now time.Time) bool {
	var timestamp string
	var signatures []string
	for _, part := range strings.Split(header, ",") {
		key, value, found := strings.Cut(strings.TrimSpace(part), "=")
		if !found {
			continue
		}
		switch key {
		case "t":
			timestamp = value
		case "v1":
			signatures = append(signatures, value)
		}
	}
	if timestamp == "" || len(signatures) == 0 {
		return false
	}
	ts, err := strconv.ParseInt(timestamp, 10, 64)
	if err != nil {
		return false
	}
	if age := now.Unix() - ts; age > toleranceSeconds || age < -toleranceSeconds {
		return false
	}

	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte(timestamp + "."))
	mac.Write(rawBody)
	expected := mac.Sum(nil)

	for _, sig := range signatures {
		received, err := hex.DecodeString(sig)
		if err == nil && hmac.Equal(received, expected) { // constant-time
			return true
		}
	}
	return false
}

func webhookHandler(w http.ResponseWriter, r *http.Request) {
	rawBody, err := io.ReadAll(http.MaxBytesReader(w, r.Body, 1<<20))
	if err != nil {
		http.Error(w, "cannot read body", http.StatusBadRequest)
		return
	}
	secret := os.Getenv("REHABIFY_WEBHOOK_SECRET") // whsec_... for this environment
	if !verifyRehabifySignature(rawBody, r.Header.Get("X-Rehabify-Signature"), secret, time.Now()) {
		http.Error(w, "invalid signature", http.StatusBadRequest)
		return
	}

	var event struct {
		ID   string `json:"id"`
		Type string `json:"type"`
	}
	if err := json.Unmarshal(rawBody, &event); err != nil {
		http.Error(w, "invalid JSON", http.StatusBadRequest)
		return
	}
	// Skip already-processed event ids, then queue the work.
	log.Printf("verified %s %s", event.ID, event.Type)
	w.WriteHeader(http.StatusOK)
}

func main() {
	http.HandleFunc("/webhooks/rehabify", webhookHandler)
	log.Fatal(http.ListenAndServe(":8080", nil))
}
```

## Common mistakes

| Symptom | Likely cause |
| :--- | :--- |
| Every signature fails | The body was parsed and re-serialized before hashing. Hash the raw bytes. In Express, use `express.raw()` on this route, before `express.json()`. |
| Sandbox works, live fails (or the other way round) | You're using the wrong environment's secret. Sandbox and live have different secrets. |
| Fails after rotating the secret | Your server still has the old secret. Deploy the new one, or temporarily accept either. |
| Intermittent timestamp failures | Your server clock is off. Sync it with NTP. The tolerance is 300 seconds. |
| `RangeError` in Node.js | You called `crypto.timingSafeEqual` with buffers of different lengths. Check the lengths first, as the sample does. |
| Signature compared with `===` or `==` | Use a constant-time comparison (`timingSafeEqual`, `hmac.compare_digest`, `hash_equals` or `hmac.Equal`). |
